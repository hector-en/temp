  done

  for f in "${files[@]}"; do
    [[ -f "$f" ]] || continue
    if grep -q $'\r' "$f" 2>/dev/null; then
      dos2unix "$f" >/dev/null 2>&1
      echo "  fixed: $f"

_lv_clip() {
  local s="$1" w="$2"
  ((${#s}<=w)) && { printf "%s" "$s"; return; }
  ((w<4)) && { printf "%s" "${s:0:w}"; return; }
  printf "%s..." "${s:0:w-3}"
}

_lv_normpath() {
  local p="$1"
  [[ -z "$p" ]] && return 0
  realpath -m "$p" 2>/dev/null || readlink -f "$p" 2>/dev/null || printf "%s" "$p"
}

_lv_shortpath() {
  local p="$1" w="${2:-24}" cb wh pnorm cbnorm whnorm
  [[ -z "$p" ]] && return 0

  cb="$(_lv_conda_base)"
  wh="$(_lv_workon_home)"

  pnorm="$(_lv_normpath "$p")"
  cbnorm="$(_lv_normpath "$cb")"
  whnorm="$(_lv_normpath "$wh")"

  if [[ -n "$cbnorm" && "$pnorm" == "$cbnorm/envs/"* ]]; then
    p="base/envs/${pnorm#"$cbnorm/envs/"}"
  elif [[ -n "$whnorm" && "$pnorm" == "$whnorm/"* ]]; then
    p="base/${pnorm#"$whnorm/"}"
  elif [[ "$pnorm" == "$HOME/"* ]]; then
    p="~/${pnorm#"$HOME/"}"
  else
    p="$pnorm"
  fi

  ((${#p}>w)) && p="...${p: -$((w-3))}"
  printf "%s" "$p"
}

_lv_capture_alias_rows() {
  local f line name body comment pending_comment
  declare -A seen=()
  local files=("$HOME/.local/bin/conda.sh" "$HOME/.local/.bash_aliases" "$HOME/.bashrc")

  for f in "${files[@]}"; do
    [[ -f "$f" ]] || continue
    pending_comment=""

    while IFS= read -r line; do
      [[ "$line" == '#!'* ]] && { pending_comment=""; continue; }

      if [[ "$line" =~ ^[[:space:]]*#[[:space:]]*(.+)$ ]]; then
        comment="${BASH_REMATCH[1]}"
        if [[ -n "$comment" && "$comment" != '>>>'* && "$comment" != '!!'* ]]; then
          pending_comment="$comment"
        else
          pending_comment=""
        fi
        continue
      fi

      [[ "$line" =~ ^[[:space:]]*$ ]] && continue

      if [[ "$line" =~ ^[[:space:]]*alias[[:space:]]+([^=[:space:]]+)= ]]; then
        name="${BASH_REMATCH[1]}"
        [[ -n "$name" && -z "${seen[$name]}" ]] || continue

        if [[ -n "$pending_comment" ]]; then
          printf "sep\t%s\t\n" "$pending_comment"
          pending_comment=""
        fi

        if alias "$name" >/dev/null 2>&1; then
          body="$(alias "$name" | sed -n "s/^alias [^=]*='\(.*\)'$/\1/p")"
          printf "data\t%s\t%s\n" "$name" "$body"
          seen["$name"]=1
        fi
        continue
      fi

      pending_comment=""
    done < "$f"
  done

  while IFS= read -r line; do
    name="${line#alias }"
    name="${name%%=*}"
    [[ -n "$name" && -z "${seen[$name]}" ]] || continue
    body="$(printf "%s\n" "$line" | sed -n "s/^alias [^=]*='\(.*\)'$/\1/p")"
    printf "data\t%s\t%s\n" "$name" "$body"
    seen["$name"]=1
  done < <(alias -p 2>/dev/null)
}

_lv_capture_conda_rows() {
  command -v conda >/dev/null 2>&1 || return 0
  local base
  base="$(conda info --base 2>/dev/null)"
  [[ -n "$base" ]] && printf "base\t%s\n" "$base"

  conda info --envs 2>/dev/null | awk '
    NF && $1 !~ /^#/ {
      n=$1
      p=($2=="*" ? $3 : $2)
      if (n!="base") print n "\t" p
    }'
}

_lv_capture_venv_rows() {
  local d w="$(_lv_workon_home)"
  declare -A seen=()

  [[ -d "$w" ]] && printf "base\t%s\n" "$w"

  if [[ -n "$VIRTUAL_ENV" && -d "$VIRTUAL_ENV" ]]; then
    printf "%s\t%s\n" "$(basename "$VIRTUAL_ENV")" "$VIRTUAL_ENV"
    seen["$VIRTUAL_ENV"]=1
  fi

  if [[ -d "$w" ]]; then
    for d in "$w"/*; do
      [[ -d "$d" ]] || continue
      [[ -n "${seen[$d]}" ]] && continue
      if [[ -f "$d/bin/activate" || -f "$d/pyvenv.cfg" ]]; then
        printf "%s\t%s\n" "$(basename "$d")" "$d"
        seen["$d"]=1
      fi
    done
  fi

  for d in "$PWD/.venv" "$PWD/venv" "$PWD/env"; do
    [[ -d "$d" ]] || continue
    [[ -n "${seen[$d]}" ]] && continue
    if [[ -f "$d/bin/activate" || -f "$d/pyvenv.cfg" ]]; then
      printf "%s\t%s\n" "$(basename "$d")" "$d"
      seen["$d"]=1
    fi
  done
}

_lv_show_important_pkgs() {
  local run_cmd="$1" p v found=0
  local pkgs=(numpy pandas scipy scikit-learn matplotlib seaborn torch tensorflow jupyter ipykernel requests fastapi flask django)
  for p in "${pkgs[@]}"; do
    v=$(eval "$run_cmd python -m pip show \"$p\" 2>/dev/null" | awk -F': ' '/^Version:/{print $2; exit}')
    [[ -n "$v" ]] && { printf "  %-14s %s\n" "$p" "$v"; found=1; }
  done
  ((found==0)) && echo "  (none of the common data/web ML packages found)"
}

_lv_print_key_envs() {
  local src="$1" line k v p
  while IFS= read -r line; do
    [[ -z "$line" ]] && continue
    k="${line%%=*}"
    v="${line#*=}"

    case "$k" in
      PATH|PYTHONPATH)
        echo "  $k:"
        IFS=':' read -r -a _lv_parts <<< "$v"
        for p in "${_lv_parts[@]}"; do
          [[ -n "$p" ]] && echo "    - $p"
        done
        ;;
      *)
        if ((${#v} > 96)); then
          echo "  $k:"
          printf "%s\n" "$v" | fold -s -w 92 | sed 's/^/    /'
        else
          echo "  $k=$v"
        fi
        ;;
    esac
  done <<< "$src"
}

_lv_inspect() {
  local kind="$1" name="$2" cpath vpath a vars

  case "$kind" in
    conda)
      cpath="$(_lv_find_conda_path "$name")"
      if [[ -n "$cpath" ]]; then
        echo -e "${_lv_bold}${_lv_green}Conda Environment:${_lv_reset} $name"
        echo "Path: $(_lv_shortpath "$cpath" 70)"
        echo "Python: $(conda run -n "$name" python -V 2>/dev/null)"
        echo "Pip:    $(conda run -n "$name" python -m pip --version 2>/dev/null)"
        echo "Key env vars:"
        vars="$(conda run -n "$name" env 2>/dev/null | grep -E '^(CONDA_PREFIX|PATH|PYTHONPATH|PIP_INDEX_URL|HTTP_PROXY|HTTPS_PROXY)=')"
        _lv_print_key_envs "$vars"
        echo "Important packages:"
        _lv_show_important_pkgs "conda run -n \"$name\""
      else
        echo "Conda env '$name' not found."
      fi
      ;;
    venv)
      vpath="$(_lv_find_venv_path "$name")"
      if [[ -n "$vpath" ]]; then
        echo -e "${_lv_bold}${_lv_cyan}Venv Environment:${_lv_reset} $name"
        echo "Path: $(_lv_shortpath "$vpath" 70)"
        (
          source "$vpath/bin/activate"
          echo "Python: $(python -V 2>/dev/null)"
          echo "Pip:    $(python -m pip --version 2>/dev/null)"
          echo "Key env vars:"
          vars="$(env | grep -E '^(VIRTUAL_ENV|PATH|PYTHONPATH|PIP_INDEX_URL|HTTP_PROXY|HTTPS_PROXY)=')"
          _lv_print_key_envs "$vars"
          echo "Important packages:"
          _lv_show_important_pkgs ""
        )
      else
        echo "Venv '$name' not found."
      fi
      ;;
    alias)
      a="$(_lv_find_alias_body "$name")"
      [[ -n "$a" ]] && echo "Alias '$name': $a" || echo "Alias '$name' not found."
      ;;
    *)
      echo "Unknown focus '$kind'. Use: alias | conda | venv"
      ;;
  esac
}

_lv_delete() {
  local kind="$1" n="$2" c v

  case "$kind" in
    conda)
      c="$(_lv_find_conda_path "$n")"
      [[ -z "$c" ]] && { echo "Conda env '$n' not found."; return; }
      echo "Will delete conda env: $n ($(_lv_shortpath "$c" 70))"
      read -r -p "Continue? [y/N] " ans
      [[ "$ans" =~ ^[Yy]$ ]] || { echo "Cancelled."; return; }
      conda env remove -n "$n" -y
      echo "Done."
      ;;
    venv)
      v="$(_lv_find_venv_path "$n")"
      [[ -z "$v" ]] && { echo "Venv '$n' not found."; return; }
      echo "Will delete venv dir: $(_lv_shortpath "$v" 70)"
      read -r -p "Continue? [y/N] " ans
      [[ "$ans" =~ ^[Yy]$ ]] || { echo "Cancelled."; return; }
      rm -rf -- "$v"
      echo "Done."
      ;;
    alias)
      if ! alias "$n" >/dev/null 2>&1; then
        echo "Alias '$n' not found."
        return
      fi
      echo "Will delete alias: $n"
      read -r -p "Continue? [y/N] " ans
      [[ "$ans" =~ ^[Yy]$ ]] || { echo "Cancelled."; return; }
      unalias "$n"
      sed -i.bak "/^[[:space:]]*alias[[:space:]]\+$n=/d" "$HOME/.local/.bash_aliases" 2>/dev/null
      echo "Done."
      ;;
    *)
      echo "Unknown focus '$kind'. Use: alias | conda | venv"
      ;;
  esac
}

_lv_alias_set() {
  local kind="$1" name="$2"
  local file header cmd alias_line tmp

  file="$(_lv_alias_edit_file)"
  header="# ${kind} aliases"
  cmd="$(_lv_alias_cmd_for_env "$kind" "$name")" || { echo "Unknown kind: $kind"; return 1; }
  alias_line="alias ${name}='${cmd}'"

  [[ -n "$file" && -f "$file" ]] || { echo "Alias target file not found."; return 1; }
  tmp="$(mktemp)" || { echo "Could not create temp file."; return 1; }

  awk -v header="$header" -v name="$name" -v alias_line="$alias_line" '
    BEGIN { in_block=0; found_header=0; added=0; exists=0; }
    {
      if ($0 == header) { found_header=1; in_block=1; print $0; next; }
      if (in_block) {
        if ($0 ~ /^#[[:space:]]+/) {
          if (!exists && !added) { print alias_line; added=1; }
          in_block=0; print $0; next;
        }
        if ($0 ~ ("^[[:space:]]*alias[[:space:]]+" name "=")) exists=1;
        print $0; next;
      }
      print $0;
    }
    END {
      if (!found_header) {
        print ""; print header; print alias_line;
      } else if (in_block && !exists && !added) {
        print alias_line;
      }
    }
  ' "$file" > "$tmp" || { rm -f "$tmp"; echo "Failed to prepare update."; return 1; }

  if cat "$tmp" > "$file"; then
    rm -f "$tmp"
    echo "Alias added/kept: $name -> $cmd"
  else
    rm -f "$tmp"
    echo "Permission denied writing alias target."
    return 1
  fi
}

_lv_alias_unset() {
  local kind="$1" name="$2" file tmp
  file="$(_lv_alias_edit_file)"
  [[ -n "$file" && -f "$file" ]] || { echo "Alias target file not found."; return 1; }

  tmp="$(mktemp)" || { echo "Could not create temp file."; return 1; }
  awk -v name="$name" '
    $0 ~ ("^[[:space:]]*alias[[:space:]]+" name "=") { next }
    { print $0 }
  ' "$file" > "$tmp" || { rm -f "$tmp"; echo "Failed to prepare update."; return 1; }

  if cat "$tmp" > "$file"; then
    rm -f "$tmp"
    echo "Alias removed (if existed): $name"
  else
    rm -f "$tmp"
    echo "Permission denied writing alias target."
    return 1
  fi
}

_lv_clone_preflight() {
  local src_kind="$1" src_name="$2" tgt_kind="$3" tgt_name="$4"
  local src_vpath="" tgt_vpath="" w=""

  command -v python3 >/dev/null 2>&1 || { echo "Missing python3. Install python3 first."; return 1; }

  if [[ "$src_kind" == "conda" || "$tgt_kind" == "conda" ]]; then
    command -v conda >/dev/null 2>&1 || {
      echo "Conda required for conda source/target. Install/initialize conda first."
      return 1
    }
  fi

  if [[ "$src_kind" == "conda" ]]; then
    [[ -n "$(_lv_find_conda_path "$src_name")" ]] || { echo "Source conda env not found: $src_name"; return 1; }
    conda run -n "$src_name" python -V >/dev/null 2>&1 || { echo "Source conda env python not usable: $src_name"; return 1; }
    conda run -n "$src_name" python -m pip --version >/dev/null 2>&1 || {
      echo "Source conda env missing pip: $src_name"
      echo "Run: conda run -n \"$src_name\" python -m ensurepip --upgrade"
      return 1
    }
  else
    src_vpath="$(_lv_find_venv_path "$src_name")"
    [[ -n "$src_vpath" && -x "$src_vpath/bin/python" ]] || { echo "Source venv python missing: $src_name"; return 1; }
    "$src_vpath/bin/python" -m pip --version >/dev/null 2>&1 || {
      echo "Source venv missing pip: $src_name"
      echo "Run: \"$src_vpath/bin/python\" -m ensurepip --upgrade"
      return 1
    }
  fi

  if [[ "$tgt_kind" == "venv" ]]; then
    w="$(_lv_workon_home)"
    mkdir -p "$w" >/dev/null 2>&1 || { echo "Cannot create WORKON_HOME: $w"; return 1; }

    tgt_vpath="$(_lv_find_venv_path "$tgt_name")"
    if [[ -n "$tgt_vpath" && -x "$tgt_vpath/bin/python" ]]; then
      "$tgt_vpath/bin/python" -m pip --version >/dev/null 2>&1 || {
        echo "Target venv exists but pip missing: $tgt_name"
        echo "Install prerequisites: sudo apt install -y python3-venv python3-pip"
        echo "Then recreate: rm -rf \"$tgt_vpath\" && python3 -m venv \"$tgt_vpath\""
        return 1
      }
    fi
  else
    if [[ -n "$(_lv_find_conda_path "$tgt_name")" ]]; then
      conda run -n "$tgt_name" python -V >/dev/null 2>&1 || { echo "Target conda env python not usable: $tgt_name"; return 1; }
      conda run -n "$tgt_name" python -m pip --version >/dev/null 2>&1 || {
        echo "Target conda env missing pip: $tgt_name"
        echo "Run: conda run -n \"$tgt_name\" python -m ensurepip --upgrade"
        return 1
      }
    fi
  fi
  return 0
}

_lv_clone_write_manifests() {
  local src_kind="$1" src_name="$2" tgt_kind="$3" tgt_name="$4" pyver="$5" wd="$6"
  shift 6
  local pkgs=("$@")

  local req="$wd/requirements.txt"
  local yml="$wd/environment.yml"
  : > "$req" || return 1

  local p pip_ver conda_ver
  local -a conda_specs=() pip_specs=()

  for p in "${pkgs[@]}"; do
    p="${p#-}"
    [[ "$p" == "jupyther" ]] && p="jupyter"
    [[ "$p" == "python" ]] && continue
    if [[ "$p" == "pip" ]]; then
      pip_specs+=("pip")
      continue
    fi
    [[ -z "$p" ]] && continue

    pip_ver=""
    conda_ver=""

    if [[ "$src_kind" == "conda" ]]; then
      conda_ver="$(conda list -n "$src_name" "$p" 2>/dev/null | awk 'NF && $1 !~ /^#/ && $1==pkg {print $2; exit}' pkg="$p")"
      pip_ver="$(conda run -n "$src_name" python -m pip show "$p" 2>/dev/null | awk -F': ' '/^Version:/{print $2; exit}')"
    else
      local sv="$(_lv_find_venv_path "$src_name")"
      [[ -n "$sv" && -x "$sv/bin/python" ]] || return 1
      pip_ver="$("$sv/bin/python" -m pip show "$p" 2>/dev/null | awk -F': ' '/^Version:/{print $2; exit}')"
    fi

    if [[ -n "$conda_ver" ]]; then
      conda_specs+=("$p=$conda_ver")
      pip_specs+=("$p==$conda_ver")
    elif [[ -n "$pip_ver" ]]; then
      pip_specs+=("$p==$pip_ver")
    else
      pip_specs+=("$p")
    fi
  done

  printf "%s\n" "${pip_specs[@]}" > "$req"

  {
    echo "name: $tgt_name"
    echo "dependencies:"
    echo "  - python=$pyver"
    echo "  - pip"
    for p in "${conda_specs[@]}"; do
      echo "  - $p"
    done
    if ((${#pip_specs[@]})); then
      echo "  - pip:"
      for p in "${pip_specs[@]}"; do
        echo "    - $p"
      done
    fi
  } > "$yml" || return 1

  printf '%s|%s\n' "$req" "$yml"
}

_lv_clone() {
  local src_kind="$1" src_name="$2"
  shift 2

  local tgt_kind="" tgt_name=""
  local -a req_pkgs=()
  local src_py="3.11"
  local src_vpath="" tgt_vpath="" workon_home=""
  local i

  [[ "$src_kind" =~ ^(conda|venv)$ ]] || { echo "Source kind must be conda|venv"; return 1; }
  [[ -n "$src_name" ]] || { echo "Source name is required."; return 1; }
  [[ "${1:-}" == "--clone" ]] || {
    echo "Usage: lv <env> <name> --clone --target <env> <name> --packages -python -numpy ..."
    return 1
  }
  shift

  while (($#)); do
    case "$1" in
      --target)
        tgt_kind="$2"
        tgt_name="$3"
        shift 3
        ;;
      --packages)
        shift
        while (($#)); do
          [[ "$1" == --* ]] && break
          req_pkgs+=("${1#-}")
          shift
        done
        ;;
      *)
        shift
        ;;
    esac
  done

  [[ "$tgt_kind" =~ ^(conda|venv)$ ]] || { echo "Target kind must be conda|venv"; return 1; }
  [[ -n "$tgt_name" ]] || { echo "Missing target env name."; return 1; }

  _lv_clone_preflight "$src_kind" "$src_name" "$tgt_kind" "$tgt_name" || return 1

  if [[ "$src_kind" == "conda" ]]; then
    src_py="$(conda run -n "$src_name" python -c 'import sys; print(f"{sys.version_info.major}.{sys.version_info.minor}")' 2>/dev/null)"
  else
    src_vpath="$(_lv_find_venv_path "$src_name")"
    [[ -n "$src_vpath" && -x "$src_vpath/bin/python" ]] || { echo "Source venv python not found: $src_name"; return 1; }
    src_py="$("$src_vpath/bin/python" -c 'import sys; print(f"{sys.version_info.major}.{sys.version_info.minor}")' 2>/dev/null)"
  fi
  [[ -n "$src_py" ]] || src_py="3.11"

  if [[ "$tgt_kind" == "conda" ]]; then
    [[ -n "$(_lv_find_conda_path "$tgt_name")" ]] || {
      conda create -n "$tgt_name" "python=$src_py" -y || { echo "Failed creating target conda env: $tgt_name"; return 1; }
    }
  else
    workon_home="$(_lv_workon_home)"
    mkdir -p "$workon_home" || { echo "Cannot create WORKON_HOME: $workon_home"; return 1; }
    tgt_vpath="$(_lv_find_venv_path "$tgt_name")"
    if [[ -z "$tgt_vpath" ]]; then
      python3 -m venv "$workon_home/$tgt_name" || { echo "Failed creating target venv: $workon_home/$tgt_name"; return 1; }
      tgt_vpath="$workon_home/$tgt_name"
    fi
    [[ -x "$tgt_vpath/bin/python" ]] || { echo "Target venv python missing: $tgt_vpath/bin/python"; return 1; }
  fi

  for i in "${!req_pkgs[@]}"; do
    [[ "${req_pkgs[$i]}" == "jupyther" ]] && req_pkgs[$i]="jupyter"
  done

  local wd out req yml tv
  wd="$(mktemp -d)" || { echo "Failed to create temp workdir."; return 1; }

  out="$(_lv_clone_write_manifests "$src_kind" "$src_name" "$tgt_kind" "$tgt_name" "$src_py" "$wd" "${req_pkgs[@]}" | tail -n 1)"
  req="${out%%|*}"
  yml="${out##*|}"

  if [[ -z "$req" || -z "$yml" || ! -f "$req" || ! -f "$yml" ]]; then
    rm -rf "$wd"
    echo "Failed to generate manifest files."
    return 1
  fi

  echo "Generated manifests:"
  echo "  requirements: $req"
  echo "  environment:  $yml"

  if [[ "$tgt_kind" == "conda" ]]; then
    conda env update -n "$tgt_name" -f "$yml" || {
      rm -rf "$wd"
      echo "conda env update failed."
      return 1
    }
  else
    tv="$(_lv_find_venv_path "$tgt_name")"
    [[ -n "$tv" && -x "$tv/bin/python" ]] || {
      rm -rf "$wd"
      echo "Target venv python not found: $tgt_name"
      return 1
    }

    _lv_pip_install_safe "$tv/bin/python" "$req" || {
      rm -rf "$wd"
      echo "pip install from requirements failed."
      return 1
    }
  fi

  rm -rf "$wd"
  echo "Clone completed: $src_kind/$src_name -> $tgt_kind/$tgt_name"
  _lv_post_clone_update_prompt "$tgt_kind" "$tgt_name"
}

_lv_post_clone_update_prompt() {
  local kind="$1" name="$2" vpath ans pkg_line
  local -a pkgs

  echo
  echo "Clone completed: $kind/$name"
  read -r -p "Post-clone update? [s]pecific packages / [o]utdated (venv) / [a]ll / [n]one: " ans

  case "${ans,,}" in
    s|specific)
      read -r -p "Enter packages (space-separated): " pkg_line
      [[ -z "$pkg_line" ]] && { echo "No packages provided. Skipping."; return 0; }
      read -r -a pkgs <<< "$pkg_line"

      if [[ "$kind" == "conda" ]]; then
        conda update -n "$name" -y "${pkgs[@]}" || {
          echo "Conda update failed, trying pip in conda env..."
          conda run -n "$name" python -m pip install -U "${pkgs[@]}"
        }
      else
        vpath="$(_lv_find_venv_path "$name")"
        [[ -n "$vpath" && -x "$vpath/bin/python" ]] || { echo "Venv not found: $name"; return 1; }
        "$vpath/bin/python" -m pip install -U "${pkgs[@]}"
      fi
      ;;
    o|outdated)
      if [[ "$kind" != "venv" ]]; then
        echo "Outdated-only flow is for venv. For conda use [a]ll or [s]pecific."
        return 0
      fi
      vpath="$(_lv_find_venv_path "$name")"
      [[ -n "$vpath" && -x "$vpath/bin/python" ]] || { echo "Venv not found: $name"; return 1; }
      "$vpath/bin/python" -m pip list --outdated --format=freeze | cut -d= -f1 | xargs -r "$vpath/bin/python" -m pip install -U
      ;;
    a|all)
      if [[ "$kind" == "conda" ]]; then
        conda update -n "$name" --all -y
      else
        vpath="$(_lv_find_venv_path "$name")"
        [[ -n "$vpath" && -x "$vpath/bin/python" ]] || { echo "Venv not found: $name"; return 1; }
        "$vpath/bin/python" -m pip list --outdated --format=freeze | cut -d= -f1 | xargs -r "$vpath/bin/python" -m pip install -U
      fi
      ;;
    ""|n|no|none)
      echo "Skipped post-clone update."
      ;;
    *)
      echo "Unknown option. Skipped post-clone update."
      ;;
  esac
}

_lv_pip_install_safe() {
  local py="$1" req="$2"
  local spec
  local -a specs=()

  [[ -x "$py" ]] || { echo "Python not executable: $py"; return 1; }
  [[ -f "$req" ]] || { echo "Requirements file not found: $req"; return 1; }

  mapfile -t specs < <(grep -E -v '^[[:space:]]*($|#)' "$req")

  for spec in "${specs[@]}"; do
    echo "Installing: $spec"
    "$py" -m pip install --progress-bar off "$spec" || {
      echo "Failed package: $spec"
      return 1
    }
  done
}

lv() {
  case "${1:-}" in
    -help|--help|-h ) _lv_help; return ;;
  esac

  if [[ -n "$1" ]]; then
    case "$1" in
      alias|conda|venv)
        if [[ -z "${2:-}" ]]; then
          echo "Usage: lv {alias|conda|venv} <name> [--clone --target <conda|venv> <name> --packages -python -pip <-pkg1 -pkg2 ...> | -new | -del]"
          return
        fi

        if [[ "${3:-}" == "--clone" ]]; then
          _lv_clone "$@"
        elif [[ "${3:-}" == "-new" || "${3:-}" == "--new" ]]; then
          _lv_create_new "$1" "$2"
        elif [[ "${3:-}" == "-del" || "${3:-}" == "--del" ]]; then
          _lv_delete "$1" "$2"
        else
          _lv_inspect "$1" "$2"
        fi
        ;;
      *)
        echo "Please specify focus: lv {alias|conda|venv} <name> [--clone --target <conda|venv> <name> --packages -python -pip <-pkg1 -pkg2 ...> | -new | -del]"
        ;;
    esac
    return
  fi

  local ip cb cb_short wb wb_short ab sep tablew
  local WA=16 WB=14 WC=10 WD=22 WE=10 WF=26
  local i max
  local kind a b c d e f
  local -a alias_rows conda_rows venv_rows

  mapfile -t alias_rows < <(_lv_capture_alias_rows)
  mapfile -t conda_rows < <(_lv_capture_conda_rows)
  mapfile -t venv_rows  < <(_lv_capture_venv_rows)

  max=${#alias_rows[@]}
  (( ${#conda_rows[@]} > max )) && max=${#conda_rows[@]}
  (( ${#venv_rows[@]}  > max )) && max=${#venv_rows[@]}

  ip="$(_lv_ip)"
  cb="$(_lv_conda_base)"
  wb="$(_lv_workon_home)"
  ab="$(_lv_alias_base)"

  tablew=$((WA+1+WB+3+WC+1+WD+3+WE+1+WF))
  sep="$(printf '%*s' "$tablew" '' | tr ' ' '-')"

  printf "Distro: ${_lv_bold}${_lv_yellow}%s${_lv_reset}   IP: ${_lv_bold}${_lv_yellow}@%s${_lv_reset}\n" "${DISTRONAME:-n/a}" "${ip:-n/a}"
  if [[ -n "$cb" ]]; then
    cb_short="${cb/#$HOME/~}"
    printf "%bConda base: %s (%s)%b\n" "${_lv_gray}" "$cb" "$cb_short" "${_lv_reset}"
  fi
  if [[ -n "$wb" ]]; then
    wb_short="${wb/#$HOME/~}"
    printf "%bVenv base:  %s (%s)%b\n" "${_lv_gray}" "$wb" "$wb_short" "${_lv_reset}"
  fi
  if [[ -n "$ab" ]]; then
    printf "%bAlias base: %s%b\n" "${_lv_gray}" "$ab" "${_lv_reset}"
  fi

  printf "${_lv_gray}%-${WA}s %-${WB}s | %-${WC}s %-${WD}s | %-${WE}s %-${WF}s${_lv_reset}\n" "ALIAS" "ACTION" "CONDA" "PATH" "VENV" "PATH"
  printf "%s\n" "$sep"

  for ((i=0; i<max; i++)); do
    kind="data"; a=""; b=""; c=""; d=""; e=""; f=""

    if (( i < ${#alias_rows[@]} )); then
      IFS=$'\t' read -r kind a b <<< "${alias_rows[i]}"
    fi
    if (( i < ${#conda_rows[@]} )); then
      IFS=$'\t' read -r c d <<< "${conda_rows[i]}"
    fi
    if (( i < ${#venv_rows[@]} )); then
      IFS=$'\t' read -r e f <<< "${venv_rows[i]}"
    fi

    d="$(_lv_shortpath "$d" "$WD")"
    f="$(_lv_shortpath "$f" "$WF")"

    if [[ "$kind" == "sep" ]]; then
      printf "%b%-${WA}s %-${WB}s${_lv_reset} | ${_lv_green}%-${WC}s %-${WD}s${_lv_reset} | ${_lv_cyan}%-${WE}s %-${WF}s${_lv_reset}\n" \
        "${_lv_gray}" "$(_lv_clip "# $a" "$WA")" "" \
        "$(_lv_clip "$c" "$WC")" "$(_lv_clip "$d" "$WD")" \
        "$(_lv_clip "$e" "$WE")" "$(_lv_clip "$f" "$WF")"
    else
      printf "%-${WA}s %-${WB}s | ${_lv_green}%-${WC}s %-${WD}s${_lv_reset} | ${_lv_cyan}%-${WE}s %-${WF}s${_lv_reset}\n" \
        "$(_lv_clip "$a" "$WA")" "$(_lv_clip "$b" "$WB")" \
        "$(_lv_clip "$c" "$WC")" "$(_lv_clip "$d" "$WD")" \
        "$(_lv_clip "$e" "$WE")" "$(_lv_clip "$f" "$WF")"
    fi
  done

  printf "%s\n" "$sep"
  echo "Try: lv -help | lv alias <name> | lv conda <name> | lv venv <name> | lv venv <name> -new | lv conda <name> -del"
  printf "%s\n" "$sep"
}
